新增组件: SubagentInlinePanel

组件层级与职责
SubagentInlinePanel 入口组件，管理收缩/展开状态
SubagentHeader 状态栏: 图标 + 任务名 + 状态 + 用时 + 工具数
SubagentCollapsedPreview 最后文本片段 + 工具标签
SubagentExpandedContent 完整消息流，复用 SDKContentBlock

Props 接口

interface SubagentInlinePanelProps {
  runId: string        // subagent run ID
  threadId: string      // 父线程 ID
  label?: string       // 任务名称 (来自 SubagentRunRecord)
  status?: SubagentRunStatus // 运行状态
  startedAt?: number     // 开始时间戳
  depth?: number       // 嵌套层级 (0-based)
}

收缩态数据计算

从 SDKMessage[] 派生收缩态数据
function deriveCollapsedData(messages: SDKMessage[]) {
  // 工具调用标签: 从 assistant 消息提取 tool_use blocks
  const toolCalls = messages
    .filter(m => m.type === 'assistant')
    .flatMap(m => m.message.content.filter(b => b.type === 'tool_use'))

  // 工具结果: 匹配 tool_use_id → 标记完成状态
  const toolResults = messages
    .filter(m => m.type === 'tool_result')

  // 最后文本片段: 最后一条 assistant 的 text block
  const lastText = messages
    .filter(m => m.type === 'assistant')
    .flatMap(m => m.message.content.filter(b => b.type === 'text'))
    .at(-1)?.text ?? ''

  return { toolCalls, toolResults, lastText }
}
