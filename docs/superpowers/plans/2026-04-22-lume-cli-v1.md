# Lume CLI V1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 构建 Lume CLI V1，支持 `status/health`、`workspaces/workspace create`、`threads/thread create/thread messages/thread send`、`files/file add`、`ask` 这条最小 headless 闭环。

**Architecture:** 先在 `@lume/sidecar` 内暴露一个窄的 headless CLI runtime facade，包装现有 workspace/thread/file/agent service；再在新的 `packages/cli` 中用 `cac` 组装命令、错误码和 JSON 输出。这样可以复用 sidecar 的持久化与运行能力，同时避免 `packages/cli` 直接跨 `rootDir` 引用 `apps/sidecar` 源码。

**Tech Stack:** TypeScript, Bun, `cac`, `zod`, `@lume/shared`, `@lume/sidecar`, `bun:test`

```ts
function normalizeRequestedSlug(slug: string | undefined): string | undefined {
  const trimmed = slug?.trim();
  if (!trimmed) {
    return undefined;
  }
  return trimmed
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/^-|-$/g, "");
}
```

2. 把创建签名改成：

```ts
export function createAgentWorkspace(
  name: string,
  options: { slug?: string } = {}
): AgentWorkspace {
  const index = readIndex();
  const existingSlugs = new Set(index.workspaces.map((workspace) => workspace.slug));
  const requestedSlug = normalizeRequestedSlug(options.slug);
  if (requestedSlug && existingSlugs.has(requestedSlug)) {
    throw new Error("workspace slug 已存在");
  }
  const slug = requestedSlug ?? slugify(name, existingSlugs);
  const now = Date.now();

  const workspace: AgentWorkspace = {
    id: randomUUID(),
    name,
    slug,
    createdAt: now,
    updatedAt: now
  };

  getAgentWorkspacePath(slug);
  ensureWorkspaceAgentAssets(slug, name);
  ensureBootstrapFiles(slug);

  index.workspaces.push(workspace);
  writeIndex(index);

  console.log(`[Agent 工作区] 已创建工作区: ${name} (slug: ${slug})`);
  return workspace;
}
```

```ts
import { existsSync } from "node:fs";
import { getConfigDir, getLumeConfigYamlPath } from "../services/infra/config-paths";
import {
  createAgentWorkspace,
  getAgentWorkspace,
  getAgentWorkspaceBySlug,
  listAgentWorkspaces
} from "../services/agent/agent-workspace-manager";
import {
  createAgentThread,
  getAgentThreadMessages,
  listAgentThreads
} from "../services/agent/agent-thread-manager";
import {
  listAgentDirectory,
  listWorkspaceDirectory,
  saveFilesToAgentSession,
  saveFilesToWorkspace
} from "../services/agent/agent-files-service";

export interface CliRuntime {
  status(): Promise<{ ok: true; runtime: "ready" }>;
  health(): Promise<{ ok: true; configDir: string; lumeConfigExists: boolean }>;
  listWorkspaces(): Promise<ReturnType<typeof listAgentWorkspaces>>;
  createWorkspace(input: { name: string; slug?: string }): Promise<ReturnType<typeof createAgentWorkspace>>;
  listThreads(input?: { workspaceSlug?: string; limit?: number }): Promise<ReturnType<typeof listAgentThreads>>;
  createThread(input: { workspaceSlug?: string; title?: string }): Promise<ReturnType<typeof createAgentThread>>;
  getThreadMessages(input: { threadId: string; limit?: number }): Promise<ReturnType<typeof getAgentThreadMessages>>;
  listFiles(input: { threadId?: string; workspaceSlug?: string }): Promise<ReturnType<typeof listAgentDirectory>>;
  addFileToThread(input: { threadId: string; sourcePath: string }): Promise<ReturnType<typeof saveFilesToAgentSession>>;
  addFileToWorkspace(input: { workspaceSlug: string; sourcePath: string }): Promise<ReturnType<typeof saveFilesToWorkspace>>;
}

function requireWorkspaceBySlug(workspaceSlug: string) {
  const workspace = getAgentWorkspaceBySlug(workspaceSlug);
  if (!workspace) throw new Error(`workspace 不存在: ${workspaceSlug}`);
  return workspace;
}

export function createCliRuntime(): CliRuntime {
  return {
    async status() {
      getConfigDir();
      return { ok: true, runtime: "ready" };
    },
    async health() {
      const configDir = getConfigDir();
      return { ok: true, configDir, lumeConfigExists: existsSync(getLumeConfigYamlPath()) };
    },
    async listWorkspaces() {
      return listAgentWorkspaces();
    },
    async createWorkspace(input) {
      return createAgentWorkspace(input.name, { slug: input.slug });
    },
    async listThreads(input = {}) {
      const workspace = input.workspaceSlug ? requireWorkspaceBySlug(input.workspaceSlug) : undefined;
      const threads = listAgentThreads().filter((item) => !workspace || item.workspaceId === workspace.id);
      return typeof input.limit === "number" ? threads.slice(0, input.limit) : threads;
    },
    async createThread(input) {
      const workspace = input.workspaceSlug ? requireWorkspaceBySlug(input.workspaceSlug) : undefined;
      return createAgentThread(input.title, undefined, workspace?.id);
    },
    async getThreadMessages(input) {
      const messages = getAgentThreadMessages(input.threadId);
      return typeof input.limit === "number" ? messages.slice(-input.limit) : messages;
    },
    async listFiles(input) {
      if (input.threadId) {
        const thread = listAgentThreads().find((item) => item.id === input.threadId);
        const workspace = thread?.workspaceId ? getAgentWorkspace(thread.workspaceId) : undefined;
        if (!thread || !workspace) throw new Error(`thread 不存在: ${input.threadId}`);
        return listAgentDirectory(workspace.slug, thread.id);
      }
      if (input.workspaceSlug) {
        requireWorkspaceBySlug(input.workspaceSlug);
        return listWorkspaceDirectory(input.workspaceSlug);
      }
      throw new Error("必须提供 --thread 或 --workspace");
    },
    async addFileToThread(input) {
      const thread = listAgentThreads().find((item) => item.id === input.threadId);
      const workspace = thread?.workspaceId ? getAgentWorkspace(thread.workspaceId) : undefined;
      if (!thread || !workspace) throw new Error(`thread 不存在: ${input.threadId}`);
      return saveFilesToAgentSession({
        workspaceSlug: workspace.slug,
        threadId: thread.id,
        files: [{ filename: input.sourcePath.split(/[/\\\\]/).pop()!, sourcePath: input.sourcePath }]
      });
    },
    async addFileToWorkspace(input) {
      requireWorkspaceBySlug(input.workspaceSlug);
      return saveFilesToWorkspace({
        workspaceSlug: input.workspaceSlug,
        files: [{ filename: input.sourcePath.split(/[/\\\\]/).pop()!, sourcePath: input.sourcePath }]
      });
    }
  };
}
```

```ts
import { CliCommandError } from "./errors";

export function writeJson(write: (value: string) => void, payload: unknown): void {
  write(`${JSON.stringify(payload, null, 2)}\n`);
}

export function writeError(write: (value: string) => void, error: unknown): number {
  if (error instanceof CliCommandError) {
    write(`${JSON.stringify({ error: { code: error.code, message: error.message } }, null, 2)}\n`);
    return error.exitCode;
  }
  write(`${JSON.stringify({ error: { code: "UNEXPECTED_ERROR", message: error instanceof Error ? error.message : String(error) } }, null, 2)}\n`);
  return 1;
}
```

在 `apps/sidecar/src/headless/cli-runtime.ts` 中新增依赖注入与 send/ask 方法。实现时在 `createCliRuntime()` 里补下面这些局部 helper 与 `return` 对象方法：

```ts
import { appendAgentMessage } from "../services/agent/agent-service";

function extractAssistantText(message: unknown): string {
  const blocks = (message as { message?: { content?: Array<{ type?: string; text?: string }> } }).message?.content ?? [];
  return blocks
    .filter((block) => block?.type === "text" && typeof block.text === "string")
    .map((block) => block.text)
    .join("\n\n")
    .trim();
}

export function createCliRuntime(
  deps: {
    appendMessage?: typeof appendAgentMessage;
  } = {}
): CliRuntime {
  const appendMessage = deps.appendMessage ?? appendAgentMessage;
  const sendThreadMessage = async (input: { threadId: string; text: string }) => {
    const chunks: string[] = [];

    return await new Promise<{
      threadId: string;
      messageId: string;
      status: "accepted";
      text: string;
    }>((resolve, reject) => {
      appendMessage(
        { threadId: input.threadId, userMessage: input.text },
        {
          onSdkMessage(message) {
            if ((message as { type?: string }).type === "assistant") {
              const text = extractAssistantText(message);
              if (text) chunks.push(text);
            }
          },
          onComplete() {
            resolve({
              threadId: input.threadId,
              messageId: `cli:${Date.now()}`,
              status: "accepted",
              text: chunks.join("\n\n").trim()
            });
          },
          onError(error) {
            reject(new Error(error));
          },
          onTitleUpdated() {},
          onAskUserQuestion() {
            reject(new Error("CLI V1 不支持交互式 ask-user-question"));
          },
          onToolPermissionRequest() {
            reject(new Error("CLI V1 不支持交互式权限确认"));
          }
        }
      );
    });
  };

  return {
    async sendThreadMessage(input) {
      return await sendThreadMessage(input);
    },
    async ask(input) {
      const threadId = input.threadId
        ?? (await this.createThread({ workspaceSlug: input.workspaceSlug })).id;
      const result = await sendThreadMessage({ threadId, text: input.text });
      return { threadId, text: result.text };
    }
  };
}
```

```ts
import { z } from "zod";

const askSchema = z.object({

    });
```
